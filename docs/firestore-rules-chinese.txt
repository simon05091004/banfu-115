rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /yearbookShirt/{docId} {
      allow read:  if docId == '115-1';
      allow write: if docId == '115-1'
        && request.resource.data.keys().hasOnly(['classes', 'layout', 'updated', 'v'])
        && request.resource.data.classes is map
        && request.resource.data.classes.size() <= 20
        && request.resource.data.layout is map
        && request.resource.data.updated is string
        && request.resource.data.updated.size() <= 40;
    }
    match /yearbookShirt/{docId}/photos/{cls} {
      allow read:  if docId == '115-1';
      allow write: if docId == '115-1'
        && request.resource.data.keys().hasOnly(['full', 'thumb', 'at', 'by'])
        && request.resource.data.full is string
        && request.resource.data.full.size() <= 760000
        && request.resource.data.thumb is string
        && request.resource.data.thumb.size() <= 120000
        && request.resource.data.at is string
        && request.resource.data.at.size() <= 40;
    }
    match /classShirt/{docId} {
      allow read:  if docId == '115-5';
      allow write: if docId == '115-5'
        && request.resource.data.keys().hasOnly(['classes', 'draws', 'quotes', 'updated', 'v'])
        && request.resource.data.classes is map
        && request.resource.data.classes.size() <= 20
        && request.resource.data.draws is list
        && request.resource.data.draws.size() <= 60
        && request.resource.data.quotes is map
        && request.resource.data.quotes.size() <= 20
        && request.resource.data.updated is string
        && request.resource.data.updated.size() <= 40;
    }
    match /classShirt/{docId}/art/{key} {
      allow read:  if docId == '115-5';
      allow write: if docId == '115-5'
        && request.resource.data.keys().hasOnly(['full', 'thumb', 'at', 'by'])
        && request.resource.data.full is string
        && request.resource.data.full.size() <= 740000
        && request.resource.data.thumb is string
        && request.resource.data.thumb.size() <= 130000
        && request.resource.data.at is string
        && request.resource.data.at.size() <= 40;
    }

    // 六年級 短除法闖關（最大公因數）
    match /mathGcd6/{docId} {
      allow read, delete: if request.auth != null;
      allow create: if request.resource.data.keys().hasOnly(
                         ['cls','seat','ts','score','stars','perfect','secs','detail'])
        && request.resource.data.cls in ['601','602','603','604','605','606',
                                         '607','608','609','610','611']
        && request.resource.data.seat is int
        && request.resource.data.seat >= 1 && request.resource.data.seat <= 40
        && request.resource.data.ts is string && request.resource.data.ts.size() <= 30
        && request.resource.data.score is int
        && request.resource.data.score >= 0 && request.resource.data.score <= 5000
        && request.resource.data.stars is int
        && request.resource.data.perfect is int
        && request.resource.data.secs is int && request.resource.data.secs <= 36000
        && request.resource.data.detail is list
        && request.resource.data.detail.size() <= 12;
      allow update: if false;
    }

    // 六年級 短除法闖關（最小公倍數）
    match /mathLcm6/{docId} {
      allow read, delete: if request.auth != null;
      allow create: if request.resource.data.keys().hasOnly(
                         ['cls','seat','ts','score','stars','perfect','secs','detail'])
        && request.resource.data.cls in ['601','602','603','604','605','606',
                                         '607','608','609','610','611']
        && request.resource.data.seat is int
        && request.resource.data.seat >= 1 && request.resource.data.seat <= 40
        && request.resource.data.ts is string && request.resource.data.ts.size() <= 30
        && request.resource.data.score is int
        && request.resource.data.score >= 0 && request.resource.data.score <= 5000
        && request.resource.data.stars is int
        && request.resource.data.perfect is int
        && request.resource.data.secs is int && request.resource.data.secs <= 36000
        && request.resource.data.detail is list
        && request.resource.data.detail.size() <= 12;
      allow update: if false;
    }
  }
}
